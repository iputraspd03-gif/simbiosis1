    /* Web Audio API Synthesizer */
    function playSound(type) {
      if (!soundEnabled) return;
      try {
        if (!audioCtx) {
          audioCtx = new (window.AudioContext || window.webkitAudioContext)();
        }
        if (audioCtx.state === 'suspended') {
          audioCtx.resume();
        }

        const osc = audioCtx.createOscillator();
        const gain = audioCtx.createGain();
        osc.connect(gain);
        gain.connect(audioCtx.destination);

        if (type === 'click') {
          osc.type = 'sine';
          osc.frequency.setValueAtTime(600, audioCtx.currentTime);
          gain.gain.setValueAtTime(0.1, audioCtx.currentTime);
          gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.05);
          osc.start();
          osc.stop(audioCtx.currentTime + 0.05);
        } else if (type === 'connect') {
          osc.type = 'triangle';
          osc.frequency.setValueAtTime(300, audioCtx.currentTime);
          osc.frequency.exponentialRampToValueAtTime(600, audioCtx.currentTime + 0.15);
          gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
          gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.15);
          osc.start();
          osc.stop(audioCtx.currentTime + 0.15);
        } else if (type === 'win') {
          osc.type = 'sine';
          osc.frequency.setValueAtTime(523.25, audioCtx.currentTime);
          osc.frequency.setValueAtTime(659.25, audioCtx.currentTime + 0.1);
          osc.frequency.setValueAtTime(783.99, audioCtx.currentTime + 0.2);
          gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
          gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.4);
          osc.start();
          osc.stop(audioCtx.currentTime + 0.4);
        }
      } catch (e) { console.log("Audio not supported"); }
    }

    /* Loading Handler */
    window.addEventListener('load', () => {
      let bar = document.getElementById('load-bar');
      let txt = document.getElementById('load-text');
      let btn = document.getElementById('start-btn');
      let progress = 0;

      let timer = setInterval(() => {
        progress += 20;
        bar.style.width = progress + '%';
        if (progress >= 100) {
          clearInterval(timer);
          txt.innerText = "Simulasi Siap!";
          btn.style.display = 'inline-block';
        }
      }, 150);
    });

    function initApp() {
      playSound('win');
      document.getElementById('splash-screen').style.display = 'none';
      updateSim();
    }

    /* Sound Button Toggle */
    document.getElementById('sound-toggle').addEventListener('click', () => {
      soundEnabled = !soundEnabled;
      const icon = document.getElementById('sound-toggle').querySelector('i');
      if (soundEnabled) {
        icon.className = 'fa-solid fa-volume-high';
      } else {
        icon.className = 'fa-solid fa-volume-xmark';
      }
    });

    /* Tab Switching Logic */
    function switchTab(tabId) {
      playSound('click');
      document.querySelectorAll('.tab-btn').forEach(btn => btn.classList.remove('active'));
      document.querySelectorAll('.tab-content').forEach(content => content.classList.remove('active'));

      event.currentTarget.classList.add('active');
      document.getElementById('tab-' + tabId).classList.add('active');
    }

    /* Simulation Control Setters */
    function setPair(pairKey) {
      playSound('click');
      simState.pair = pairKey;
      document.querySelectorAll('#pair-bee, #pair-bird, #pair-clown, #pair-ant').forEach(b => b.classList.remove('active'));
      document.getElementById('pair-' + pairKey).classList.add('active');
      updateSim();
    }

    function toggleInteraction() {
      simState.isConnected = !simState.isConnected;
      playSound('connect');
      const btn = document.getElementById('interact-btn');

      if (simState.isConnected) {
        btn.className = 'action-toggle-btn on';
        btn.innerHTML = '<i class="fa-solid fa-unlink"></i> LEPAS INTERAKSI';
      } else {
        btn.className = 'action-toggle-btn off';
        btn.innerHTML = '<i class="fa-solid fa-link"></i> HUBUNGKAN ORGANISME';
      }
      updateSim();
    }

    function resetSim() {
      playSound('click');
      simState.isConnected = false;
      simState.paramVal = 50;
      document.getElementById('param-slider').value = 50;
      const btn = document.getElementById('interact-btn');
      btn.className = 'action-toggle-btn off';
      btn.innerHTML = '<i class="fa-solid fa-link"></i> HUBUNGKAN ORGANISME';
      setPair('bee');
    }

    /* Update Simulation Graphics & Logic */
    function updateSim() {
      const sliderVal = document.getElementById('param-slider').value;
      simState.paramVal = sliderVal;
      document.getElementById('param-val').innerText = sliderVal + '%';

      const data = pairData[simState.pair];

      // Update Icons and Labels
      document.getElementById('icon-a').innerText = data.iconA;
      document.getElementById('label-a').innerText = data.labelA;
      document.getElementById('icon-b').innerText = data.iconB;
      document.getElementById('label-b').innerText = data.labelB;

      const arrow = document.getElementById('arrow');
      const orgAIcon = document.getElementById('icon-a');
      const orgBIcon = document.getElementById('icon-b');
      const desc = document.getElementById('status-desc');

      if (simState.isConnected) {
        arrow.classList.add('active');
        document.getElementById('benefit-a').innerText = data.benefitAText;
        document.getElementById('benefit-b').innerText = data.benefitBText;

        // Animated movement towards center based on slider value
        const shiftX = (sliderVal / 100) * 30;
        orgAIcon.style.transform = `translateX(${shiftX}px) scale(1.1)`;
        orgBIcon.style.transform = `translateX(-${shiftX}px) scale(1.1)`;

        // Calculate Meters Output
        document.getElementById('meter-a').innerText = `+${sliderVal}%`;
        document.getElementById('meter-b').innerText = `+${sliderVal}%`;

        desc.innerHTML = `<b>Simbiosis Mutualisme Aktif:</b> ${data.desc}`;
        desc.style.color = '#a7f3d0';
      } else {
        arrow.classList.remove('active');
        document.getElementById('benefit-a').innerText = 'Belum berinteraksi';
        document.getElementById('benefit-b').innerText = 'Belum berinteraksi';

        orgAIcon.style.transform = 'translateX(0) scale(1)';
        orgBIcon.style.transform = 'translateX(0) scale(1)';

        document.getElementById('meter-a').innerText = '+0%';
        document.getElementById('meter-b').innerText = '+0%';

        desc.innerHTML = 'Klik <b>HUBUNGKAN ORGANISME</b> untuk memulai interaksi simbiosis mutualisme!';
        desc.style.color = '#94a3b8';
      }
    }
